      const prepared = await prepareImageVariantsMeasured(sourceFile, {
        signal: current.controller.signal,
        onProgress: (progress) => {
          if (current !== session) return;
          const percent = Math.max(0, Math.min(100, Math.round(Number(progress) || 0)));
          status.textContent = `第 ${index + 1}/${current.items.length} 张：正在生成高清图和列表图 ${percent}%`;
        }
      });
      if (current !== session) return;
      item.uploadPromise = uploadPreparedImagePair(prepared, {
        taskId: task.id,
        businessType: 'member-checkin',
        onStage: (stage) => {
          if (current === session) status.textContent = `第 ${index + 1}/${current.items.length} 张：${stage}`;
        }
      }, current.controller.signal);
      updateReadyState();
      const pair = await item.uploadPromise;
      if (current !== session) return;
      item.compressed = prepared.display;
      item.mediaId = pair.display.mediaId;
      item.thumbMediaId = pair.thumb.mediaId;
