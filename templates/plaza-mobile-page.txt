const plazaLayoutTunerEnabled = () => {
  try {
    return new URLSearchParams(location.search).get('layoutDebug') === '1';
  } catch {
    return false;
  }
};

const defaultPlazaLayoutConfig = () => Object.fromEntries(
  PLAZA_LAYOUT_TUNER_FIELDS.map((field) => [field.key, field.defaultValue])
);

const normalizePlazaLayoutConfig = (candidate = {}) => Object.fromEntries(
  PLAZA_LAYOUT_TUNER_FIELDS.map((field) => {
    const parsed = Number(candidate[field.key]);
    const value = Number.isFinite(parsed) ? parsed : field.defaultValue;
    const clamped = Math.min(field.max, Math.max(field.min, value));
    const decimals = String(field.step).includes('.') ? String(field.step).split('.')[1].length : 0;
    return [field.key, Number(clamped.toFixed(decimals))];
  })
);

const loadPlazaLayoutConfig = () => {
  try {
    return normalizePlazaLayoutConfig(JSON.parse(localStorage.getItem(PLAZA_LAYOUT_TUNER_KEY) || '{}'));
  } catch {
    return defaultPlazaLayoutConfig();
  }
};

const clearPlazaLayoutTunerStyles = () => {
  PLAZA_LAYOUT_TUNER_FIELDS.forEach((field) => document.body.style.removeProperty(field.css));
};

const readPlazaLayoutNumber = (cssName, fallback) => {
  const value = Number.parseFloat(getComputedStyle(document.body).getPropertyValue(cssName));
  return Number.isFinite(value) ? value : fallback;
};

const applyPlazaLayoutConfig = (candidate, options = {}) => {
  const config = normalizePlazaLayoutConfig(candidate);
  PLAZA_LAYOUT_TUNER_FIELDS.forEach((field) => {
    document.body.style.setProperty(field.css, `${config[field.key]}${field.unit}`);
  });
  if (options.persist) {
    try { localStorage.setItem(PLAZA_LAYOUT_TUNER_KEY, JSON.stringify(config)); } catch {}
  }
  requestAnimationFrame(() => {
    document.querySelectorAll('.plaza-card-cover img').forEach((image) => {
      if (image.complete && image.naturalWidth) applyPlazaCoverRatio(image);
    });
    rebalancePlazaColumns();
  });
  return config;
};

const copyPlazaLayoutConfig = async (config) => {
  const payload = JSON.stringify({ version: 1, mode: 'plaza-layout', values: normalizePlazaLayoutConfig(config) }, null, 2);
  try {
    await navigator.clipboard.writeText(payload);
  } catch {
    const textarea = document.createElement('textarea');
    textarea.value = payload;
    textarea.style.position = 'fixed';
    textarea.style.opacity = '0';
    document.body.append(textarea);
    textarea.select();
    document.execCommand('copy');
    textarea.remove();
  }
  if (typeof showToast === 'function') showToast('布局参数已复制');
};

const installPlazaLayoutTuner = () => {
  const existingButton = document.querySelector('.plaza-layout-tuner-button');
  const existingPanel = document.querySelector('.plaza-layout-tuner');
  if (!plazaLayoutTunerEnabled()) {
    clearPlazaLayoutTunerStyles();
    existingButton?.remove();
    existingPanel?.remove();
    return;
  }

  let currentConfig = applyPlazaLayoutConfig(loadPlazaLayoutConfig());
  if (existingButton && existingPanel) return;

  const button = document.createElement('button');
  button.type = 'button';
  button.className = 'plaza-layout-tuner-button';
  button.textContent = '调布局';
  button.setAttribute('aria-controls', 'plazaLayoutTuner');
  button.setAttribute('aria-expanded', 'false');

  const panel = document.createElement('section');
  panel.id = 'plazaLayoutTuner';
  panel.className = 'plaza-layout-tuner';
  panel.hidden = true;
  panel.innerHTML = `
    <div class="plaza-layout-tuner-header">
      <div><strong>活动广场布局调试</strong><br><span>只影响当前浏览器，不会直接改正式样式</span></div>
      <button class="plaza-layout-tuner-close" type="button" aria-label="关闭布局面板">×</button>
    </div>
    <div class="plaza-layout-tuner-grid">
      ${PLAZA_LAYOUT_TUNER_FIELDS.map((field) => `
        <div class="plaza-layout-control">
          <label for="plazaLayout_${field.key}">${field.label}</label>
          <input id="plazaLayout_${field.key}" type="range" min="${field.min}" max="${field.max}" step="${field.step}" value="${currentConfig[field.key]}" data-layout-key="${field.key}">
          <output for="plazaLayout_${field.key}" data-layout-output="${field.key}">${currentConfig[field.key]}${field.unit}</output>
        </div>`).join('')}
    </div>
    <div class="plaza-layout-tuner-actions">
      <button type="button" class="secondary" data-layout-reset>恢复默认</button>
      <button type="button" data-layout-copy>复制配置</button>
      <button type="button" class="secondary" data-layout-exit>退出调试</button>
    </div>`;

  const syncControls = () => {
    PLAZA_LAYOUT_TUNER_FIELDS.forEach((field) => {
      const input = panel.querySelector(`[data-layout-key="${field.key}"]`);
      const output = panel.querySelector(`[data-layout-output="${field.key}"]`);
      if (input) input.value = String(currentConfig[field.key]);
      if (output) output.textContent = `${currentConfig[field.key]}${field.unit}`;
    });
  };
  const closePanel = () => {
    panel.hidden = true;
    button.setAttribute('aria-expanded', 'false');
  };

  button.onclick = () => {
    panel.hidden = !panel.hidden;
    button.setAttribute('aria-expanded', String(!panel.hidden));
  };
  panel.querySelector('.plaza-layout-tuner-close').onclick = closePanel;
  panel.querySelectorAll('[data-layout-key]').forEach((input) => {
    input.oninput = () => {
      currentConfig = applyPlazaLayoutConfig({ ...currentConfig, [input.dataset.layoutKey]: Number(input.value) }, { persist: true });
      syncControls();
    };
  });
  panel.querySelector('[data-layout-reset]').onclick = () => {
    try { localStorage.removeItem(PLAZA_LAYOUT_TUNER_KEY); } catch {}
    currentConfig = applyPlazaLayoutConfig(defaultPlazaLayoutConfig());
    syncControls();
    if (typeof showToast === 'function') showToast('已恢复默认布局');
  };
  panel.querySelector('[data-layout-copy]').onclick = () => { void copyPlazaLayoutConfig(currentConfig); };
  panel.querySelector('[data-layout-exit]').onclick = () => {
    const url = new URL(location.href);
    url.searchParams.delete('layoutDebug');
    location.href = url.toString();
  };
  document.body.append(button, panel);
};

const rebalancePlazaColumns = () => {
  const grid = document.querySelector('.plaza-grid');
  const columns = [...(grid?.querySelectorAll('[data-plaza-column]') || [])];
  if (!grid || columns.length !== 2 || grid.dataset.rebalancing === 'true') return;
  const cards = [...grid.querySelectorAll('.plaza-card')]
    .sort((left, right) => Number(left.dataset.cardIndex || 0) - Number(right.dataset.cardIndex || 0));
  if (!cards.length) return;
  grid.dataset.rebalancing = 'true';
  columns.forEach((column) => column.replaceChildren());
  cards.forEach((card, index) => {
    const target = index < 2
      ? columns[index]
      : columns[0].getBoundingClientRect().height <= columns[1].getBoundingClientRect().height
        ? columns[0]
        : columns[1];
    target.append(card);
  });
  grid.dataset.rebalancing = 'false';
};

const applyPlazaCoverRatio = (image) => {
  const shell = image?.closest?.('.plaza-card-cover');
  if (!shell) return;
  const naturalRatio = Number(image.naturalWidth) / Number(image.naturalHeight);
  const defaultFeedRatio = Math.min(4 / 3, Math.max(3 / 4, naturalRatio));
  const minRatio = readPlazaLayoutNumber('--plaza-cover-min-ratio', 3 / 4);
  const maxRatio = Math.max(minRatio, readPlazaLayoutNumber('--plaza-cover-max-ratio', 4 / 3));
  const feedRatio = Number.isFinite(naturalRatio) && naturalRatio > 0
    ? Math.min(maxRatio, Math.max(minRatio, naturalRatio))
    : Number.isFinite(defaultFeedRatio) ? defaultFeedRatio : 4 / 3;
  shell.style.aspectRatio = String(feedRatio);
  shell.dataset.feedRatio = feedRatio.toFixed(3);
  shell.classList.add('loaded');
  requestAnimationFrame(rebalancePlazaColumns);
};

async function plaza(sort = 'latest', page = 1, month = '', query = '') {
  const pageEpoch = beginNavigation();
  const safeSort = sort === 'hot' ? 'hot' : 'latest';
  const safeQuery = String(query || '').trim().slice(0, 40);
  const cacheKey = scopedCacheKey('plaza', safeSort, page, safeQuery);
  const cached = readViewCache(plazaViewCache, cacheKey);
  const path = `/api/plaza?sort=${safeSort}&page=${page}&limit=20${safeQuery ? `&q=${encodeURIComponent(safeQuery)}` : ''}`;
  if (cached) {
    renderPlazaPage(cached.data, safeSort, page, '', pageEpoch, { query: safeQuery });
    const refresh = async () => {
      try {
        const result = await api(path);
        writeViewCache(plazaViewCache, cacheKey, result);
        if (!isCurrentNavigation(pageEpoch)
            || document.body.dataset.view !== 'plaza'
            || document.querySelector('.plaza-detail')) return;
        renderPlazaPage(result, safeSort, page, '', pageEpoch, { preserveScroll: true, query: safeQuery });
      } catch {
        if (!isCurrentNavigation(pageEpoch)) return;
        const status = document.querySelector('#viewCacheStatus');
        if (status) {
          status.hidden = false;
          status.textContent = '当前显示的是已缓存内容，最新数据刷新失败。';
        }
      }
    };
    if (cacheIsFresh(cached)) queueMicrotask(() => { void refresh(); });
    else void refresh();
    return;
  }
  const result = await api(path);
  writeViewCache(plazaViewCache, cacheKey, result);
  renderPlazaPage(result, safeSort, page, '', pageEpoch, { query: safeQuery });
}
