(() => {
  let preservedHealthSection = null;
  let preservedHealthDate = '';

  const rememberHealthSection = () => {
    const section = document.querySelector('#activityTasks');
    const date = studentViewState?.data?.date || '';
    if (user?.role === 'student'
        && user.trackId === 'health'
        && date
        && section?.querySelector('[data-health-client-slot]')) {
      preservedHealthSection = section;
      preservedHealthDate = date;
    }
  };

  app.addEventListener('click', (event) => {
    if (event.target?.closest?.('[data-health-client-slot]')) rememberHealthSection();
  }, true);

  const previousStableHealthStudent = student;
  student = async function healthClientStableStudent(...args) {
    rememberHealthSection();
    const nextDate = args[0]?.date || studentViewState?.data?.date || '';
    const preserved = user?.role === 'student'
      && user.trackId === 'health'
      && preservedHealthDate === nextDate
      && preservedHealthSection?.querySelector('[data-health-client-slot]')
      ? preservedHealthSection
      : null;
    const result = await previousStableHealthStudent(...args);
    const replacement = document.querySelector('#activityTasks');
    if (preserved && replacement && replacement !== preserved && app.contains(replacement)) {
      replacement.replaceWith(preserved);
      queueMicrotask(() => window.dispatchEvent(new Event('focus')));
    }
    return result;
  };
})();
